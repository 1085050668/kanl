local function getNearbyPlayers(range)
    local nearby = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local targetHRP = player.Character:FindFirstChild("HumanoidRootPart")
            if targetHRP then
                local dist = (HumanoidRootPart.Position - targetHRP.Position).Magnitude
                if dist <= range then
                    table.insert(nearby, player)
                end
            end
        end
    end
    return nearby
end

local function getClosestPlayer()
    local closest = nil
    local minDist = math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local targetHRP = player.Character:FindFirstChild("HumanoidRootPart")
            if targetHRP then
                local dist = (HumanoidRootPart.Position - targetHRP.Position).Magnitude
                if dist < minDist then
                    minDist = dist
                    closest = player
                end
            end
        end
    end
    return closest, minDist
end

local function useSkill(skillName, target)
    local tool = Backpack:FindFirstChild(skillName) or Character:FindFirstChild(skillName)
    if tool then
        Humanoid:EquipTool(tool)
        if target and target.Character then
            local targetHRP = target.Character:FindFirstChild("HumanoidRootPart")
            if targetHRP then
                HumanoidRootPart.CFrame = CFrame.new(HumanoidRootPart.Position, targetHRP.Position)
            end
        end
        tool:Activate()
    end
end

local function autoAttack()
    local target, dist = getClosestPlayer()
    if target and dist <= 50 then
        for skillName, data in pairs(skillData) do
            for animId, cooldown in pairs(data) do
                if tick() - (lastSkillUse[animId] or 0) >= cooldown then
                    useSkill(skillName, target)
                    lastSkillUse[animId] = tick()
                    break
                end
            end
        end
    end
end

local function autoFarm()
    local target, dist = getClosestPlayer()
    if target and dist <= attackRange then
        local targetHRP = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
        if targetHRP then
            local targetHumanoid = target.Character:FindFirstChild("Humanoid")
            if targetHumanoid and targetHumanoid.Health > 0 then
                HumanoidRootPart.CFrame = targetHRP.CFrame * CFrame.new(0, 0, 3)
                autoAttack()
            end
        end
    end
end

local function getAnimationId(animTrack)
    if animTrack and animTrack.Animation then
        return animTrack.Animation.AnimationId
    end
    return nil
end

local function isSkillAnimation(animId)
    for skillName, data in pairs(skillData) do
        for id, _ in pairs(data) do
            if id == animId then
                return true
            end
        end
    end
    return false
end

RunService.Heartbeat:Connect(function()
    if autoDodgeEnabled then
        local target, dist = getClosestPlayer()
        if target and dist <= 15 then
            local targetHRP = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
            if targetHRP then
                local targetHumanoid = target.Character:FindFirstChild("Humanoid")
                if targetHumanoid then
                    local animTracks = targetHumanoid:GetPlayingAnimationTracks()
                    for _, track in ipairs(animTracks) do
                        local animId = getAnimationId(track)
                        if animId and isSkillAnimation(animId) then
                            local dodgeDirection = (HumanoidRootPart.Position - targetHRP.Position).unit
                            HumanoidRootPart.CFrame = HumanoidRootPart.CFrame + dodgeDirection * 20
                            break
                        end
                    end
                end
            end
        end
    end
end)
